@app.route("/update_cart", methods=["POST"]) @login_required def update_cart(): data = request.json product_id = data["id"] action = data["action"] item = Cart.query.filter_by( user_id=current_user.id, product_id=product_id ).first() if not item: return jsonify({"status": "error"}) if action == "increase": item.quantity += 1 elif action == "decrease" and item.quantity > 1: item.quantity -= 1 db.session.commit() product = Product.query.get(product_id) subtotal = product.price * item.quantity # Calculate new total items = Cart.query.filter_by(user_id=current_user.id).all() total = sum( Product.query.get(i.product_id).price * i.quantity for i in items ) return jsonify({ "status": "updated", "quantity": item.quantity, "subtotal": subtotal, "total": total })
